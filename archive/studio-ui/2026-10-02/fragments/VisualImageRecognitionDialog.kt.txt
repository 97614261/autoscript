@Composable
private fun RecognitionPreviewDialog(preview: RecognitionPreview, initialBounds: List<String>?, onDismiss: () -> Unit, onRegion: (List<String>) -> Unit) {
    var bounds by remember(preview, initialBounds) {
        mutableStateOf(initialBounds?.takeIf { values ->
            imageRecognitionBounds(values)?.let { it.get("right").asInt <= preview.width && it.get("bottom").asInt <= preview.height } == true
        })
    }
    var dragStart by remember { mutableStateOf(Offset.Zero) }
    var dragEnd by remember { mutableStateOf(Offset.Zero) }
    Dialog(onDismissRequest = onDismiss, properties = DialogProperties(usePlatformDefaultWidth = false)) {
        Surface(Modifier.fillMaxWidth(.94f).widthIn(max = 620.dp).fillMaxHeight(.82f), color = Color.White) {
            Column {
                RecognitionHeader(if (initialBounds == null) "模板预览 · ${preview.width} × ${preview.height}px" else "拖动框选范围 · 原图像素", onDismiss)
                BoxWithConstraints(Modifier.weight(1f).fillMaxWidth().background(Color(0xFF20242C)), contentAlignment = Alignment.Center) {
                    val scale = min(constraints.maxWidth.toFloat() / preview.width, constraints.maxHeight.toFloat() / preview.height)
                    val density = LocalDensity.current
                    val widthDp = with(density) { (preview.width * scale).toDp() }
                    val heightDp = with(density) { (preview.height * scale).toDp() }
                    Box(Modifier.size(widthDp, heightDp)) {
                        Image(preview.bitmap.asImageBitmap(), "原图", Modifier.fillMaxSize(), contentScale = ContentScale.FillBounds)
                        Canvas(Modifier.fillMaxSize().pointerInput(preview, initialBounds != null, scale) {
                            if (initialBounds != null && scale > 0) detectDragGestures(
                                onDragStart = { dragStart = it; dragEnd = it },
                                onDrag = { change, amount ->
                                    change.consume()
                                    dragEnd += amount
                                    bounds = imageRecognitionDragRegion(dragStart.x / scale, dragStart.y / scale, dragEnd.x / scale, dragEnd.y / scale, preview.width, preview.height)
                                },
                            )
                        }) {
                            bounds?.let(::imageRecognitionBounds)?.let { region ->
                                drawRect(Color(0xFF00E5FF), Offset(region.get("left").asFloat * scale, region.get("top").asFloat * scale),
                                    Size((region.get("right").asFloat - region.get("left").asFloat) * scale, (region.get("bottom").asFloat - region.get("top").asFloat) * scale), style = Stroke(2.dp.toPx()))
                            }
                        }
                    }
                }
                if (initialBounds != null) Text("X/Y/宽/高：${bounds?.joinToString(", ") ?: "请拖动选框"}", Modifier.padding(8.dp), fontSize = 12.sp)
                RecognitionFooter("取消", if (initialBounds == null) "关闭" else "使用范围", onDismiss) {
                    if (initialBounds == null) onDismiss() else bounds?.takeIf { imageRecognitionBounds(it) != null }?.let(onRegion)
                }
            }
        }
    }
}
